import { createHash } from "node:crypto";
import { existsSync, readFileSync, writeFileSync } from "node:fs";
import { createRequire } from "node:module";
import { join } from "node:path";
import { pathToFileURL } from "node:url";
import { readTraceEvidence, traceCaseFailure, traceEvidenceErrorReceipt } from "./trace-evidence-reader.mjs";

const [root, cachePath, outputPath] = process.argv.slice(2);
const variant = process.env.AB_VARIANT ?? "unknown";
const sourcePin = process.env.AB_SOURCE_PIN ?? "e21d54d5643df95f9bb12d04a7d7a7988a831883";
const sourceArchiveSha = process.env.AB_SOURCE_ARCHIVE_SHA ?? "";
const patchSha = process.env.AB_PATCH_SHA ?? "";
const expectedCacheSha = process.env.AB_CACHE_SHA ?? "3dead372e3c079f968979908c9bb5434fa68298b1cf75814de30f706a1ae1699";
const fixtureShaExpected = "b614dd7d72e54bc7faf1ec97f2a72f5edfe67238825c934704c9930048439a7a";
const focusFixtureIds = [
  "dirty-paths-are-owned",
  "wave-sizing-for-head-sessions",
  "isolation-skippy-query-cannot-reach-bilby",
  "isolation-bilby-query-cannot-reach-rico",
  "isolation-shared-kb-query-cannot-reach-skippy",
];
const fixtureSourceLabels = {
  thoughts: "thought",
  decisions: "decision",
  relationships: "relationship",
  projects: "project",
  sessions: "session",
  ob_session_events: "session_event",
};
const limit = 10;
const offset = 0;
const namespace = "rico";

function sha256(value) {
  return createHash("sha256").update(value).digest("hex");
}
function record(value) {
  return value && typeof value === "object" ? value : {};
}
function rowKey(row) {
  const value = record(row);
  const identity = value.row_id ?? value.id ?? value.path;
  return `${typeof value.source_type === "string" ? value.source_type : "?"}:${typeof identity === "string" ? identity : "?"}`;
}
function fixtureRefKey(ref) {
  const source = fixtureSourceLabels[ref.source_type] ?? ref.source_type;
  return `${source}:${ref.id}`;
}
function spanByName(body, name) {
  return (Array.isArray(body?.spans) ? body.spans : []).find((span) => span?.name === name);
}
function duration(span) {
  const value = record(span?.metadata).duration_ms;
  return typeof value === "number" ? value : null;
}
function armRows(evidence, kind) {
  return evidence.candidates.map((candidate, index) => {
    const row = record(candidate);
    const score = kind === "vector"
      ? (typeof row.similarity === "number" ? row.similarity : row.distance ?? null)
      : (typeof row.bm25_score === "number" ? row.bm25_score : null);
    return { rank: index + 1, id: rowKey(row), namespace: row.namespace ?? null, score };
  });
}
function fusionRows(evidence) {
  return evidence.candidates.map((candidate, index) => {
    const row = record(candidate);
    return {
      rank: index + 1,
      id: rowKey(row),
      namespace: row.namespace ?? null,
      score: typeof row.rrf_score === "number" ? row.rrf_score : null,
      chosen: row.chosen === true,
    };
  });
}
function targetEvidence(rows, key) {
  const found = rows.find((row) => row.id === key);
  return { present: Boolean(found), rank: found?.rank ?? null, score: found?.score ?? found?.rrf_score ?? null };
}
function quietLogger() {
  return { info: () => undefined, warn: () => undefined };
}
function writeOutput(value) {
  writeFileSync(outputPath, `${JSON.stringify(value, null, 2)}\n`, "utf8");
}
function fail(message) {
  writeOutput({
    schema: "openbrain.recall_ab_trace.v1",
    status: "HARNESS_ERROR",
    variant,
    source_head: sourcePin,
    error_class: message,
    cases: [],
  });
  process.stderr.write(`HARNESS_ERROR ${message}\n`);
  process.exit(3);
}
if (!root || !cachePath || !outputPath || !existsSync(cachePath)) fail("missing_arguments_or_cache");

const fixtureRaw = readFileSync(join(root, "eval/recall/fixtures.json"), "utf8");
const fixtureSha = sha256(fixtureRaw);
const fixtureDocument = JSON.parse(fixtureRaw);
if (fixtureSha !== fixtureShaExpected || fixtureDocument.fixtures?.length !== 24) fail("fixture_identity_drift");
const fixtures = fixtureDocument.fixtures;
const byId = new Map(fixtures.map((fixture) => [fixture.id, fixture]));
const cacheRaw = readFileSync(cachePath, "utf8");
if (sha256(cacheRaw) !== expectedCacheSha) fail("combined_cache_hash_drift");
const cache = record(JSON.parse(cacheRaw));
const cacheEntries = record(cache.results);
for (const fixture of fixtures) {
  const entry = record(cacheEntries[fixture.id]);
  if (entry.query_sha256 !== sha256(fixture.query) || !Array.isArray(entry.embedding) ||
      entry.embedding.length !== 768 || entry.embedding.some((value) => typeof value !== "number" || !Number.isFinite(value))) {
    fail(`cache_entry_invalid:${fixture.id}`);
  }
}

const require = createRequire(join(root, "package.json"));
const { Pool } = require("pg");
const pino = require("pino");
const { installMcpTracing } = await import(pathToFileURL(join(root, "server/observability/langfuse-tracing.ts")).href);
const { executeSearch } = await import(pathToFileURL(join(root, "server/tools/search-engine.ts")).href);
const { ALL_TABLES } = await import(pathToFileURL(join(root, "server/tools/search-constants.ts")).href);
const pool = new Pool({
  host: process.env.PGHOST,
  port: Number(process.env.PGPORT ?? "5432"),
  database: process.env.PGDATABASE,
  user: process.env.PGUSER,
  ...(process.env.PGPASSWORD === undefined ? {} : { password: process.env.PGPASSWORD }),
  max: 2,
  options: "-c default_transaction_read_only=on",
});
const traces = new Map();
const sink = {
  emit(body) {
    const fixtureId = typeof body?.input?.fixture_id === "string" ? body.input.fixture_id : null;
    if (fixtureId) traces.set(fixtureId, body);
  },
  forceFlush: () => Promise.resolve(),
  shutdown: () => Promise.resolve(),
};
const tracedServer = {
  handlers: new Map(),
  registerTool(name, _config, callback) { this.handlers.set(name, callback); },
};
installMcpTracing(tracedServer, {
  config: { enabled: true, maskingEnabled: true, endpoint: "http://127.0.0.1:1", publicKey: "recall-ab", secretKey: "recall-ab" },
  logger: quietLogger(),
  sink,
});
const sourceTables = [...ALL_TABLES, "session_events"];
tracedServer.registerTool("recall_ab_trace", { inputSchema: {} }, async (args) => {
  const fixture = byId.get(args.fixture_id);
  if (!fixture) throw new Error("fixture_not_found");
  const embedding = cacheEntries[args.fixture_id]?.embedding;
  return executeSearch(
    { pool, embedFn: async () => embedding, logger: pino({ level: "silent" }) },
    sourceTables, fixture.query, limit, "hybrid", undefined, offset, fixture.namespace, {},
  );
});

const result = {
  schema: "openbrain.recall_ab_trace.v1",
  status: "captured",
  variant,
  source_head: sourcePin,
  source_archive_sha256: sourceArchiveSha,
  patch_sha256: patchSha || null,
  fixture_sha256: fixtureSha,
  fixture_count: fixtures.length,
  focus_fixture_ids: focusFixtureIds,
  namespace,
  mode: "hybrid",
  limit,
  offset,
  cache_sha256: sha256(cacheRaw),
  model: cache.model,
  embedding_dimensions: cache.dimensions,
  model_calls: { embedding: 0, chat: 0, grader: 0, query_expansion: 0, rerank: 0 },
  cases: [],
};
const handler = tracedServer.handlers.get("recall_ab_trace");
try {
  for (const fixture of fixtures) {
    traces.delete(fixture.id);
    const started = performance.now();
    try {
      await handler({ fixture_id: fixture.id }, { authInfo: { role: "admin", clientId: "rico", tokenClientId: "rico", agentId: "recall-ab", namespaceSource: "header" } });
      const body = traces.get(fixture.id);
      if (!body) throw new Error("missing_trace");
      const vectorSpan = spanByName(body, "retrieval.vector_query");
      const lexicalSpan = spanByName(body, "retrieval.keyword_query");
      const rankSpan = spanByName(body, "retrieval.rank_rrf");
      const executeSpan = spanByName(body, "retrieval.execute");
      const vectorEvidence = readTraceEvidence(vectorSpan, ["similarity", "distance"]);
      const lexicalEvidence = readTraceEvidence(lexicalSpan, ["bm25_score"]);
      const rankEvidence = readTraceEvidence(rankSpan, ["rrf_score"]);
      const executeEvidence = readTraceEvidence(executeSpan, ["rrf_score", "similarity", "bm25_score", "usefulness"]);
      const evidenceByArm = { vector: vectorEvidence, lexical: lexicalEvidence, rrf: rankEvidence, execute: executeEvidence };
      const receiptEvidence = (evidence) => ({
        payload_degraded: evidence.payload_degraded,
        payload_degradation_reason: evidence.payload_degradation_reason,
        additional_spans_omitted: evidence.additional_spans_omitted,
        payload_evidence_truncated: evidence.payload_evidence_truncated,
        span_present: evidence.span_present,
        declared_count: evidence.declared_count,
        evidence_complete: evidence.evidence_complete,
        candidate_shape_complete: evidence.candidate_shape_complete,
        count_complete: evidence.count_complete,
      });
      const vectorRows = armRows(vectorEvidence, "vector");
      const lexicalRows = armRows(lexicalEvidence, "lexical");
      const fusedRows = fusionRows(rankEvidence);
      const selectedRows = fusedRows.filter((row) => row.chosen).map((row, index) => ({ ...row, selected_rank: index + 1 }));
      const fusionById = new Map(fusedRows.map((row) => [row.id, row]));
      const actualRows = executeEvidence.candidates.map((candidate, index) => {
        const id = rowKey(candidate);
        const fused = fusionById.get(id);
        return {
          rank: index + 1,
          id,
          namespace: candidate.namespace ?? fused?.namespace ?? null,
          rrf_score: fused?.score ?? null,
          fused_rank: fused?.rank ?? null,
        };
      });
      const expected = fixture.expected ?? [];
      const forbidden = fixture.forbidden ?? [];
      const expectedKeys = expected.filter((ref) => ref.label === 2).map(fixtureRefKey);
      const forbiddenKeys = forbidden.map(fixtureRefKey);
      const allIds = actualRows.map((row) => row.id);
      const evidenceFailure = traceCaseFailure(evidenceByArm, fixture.isolation === true, actualRows);
      if (evidenceFailure) {
        result.status = "TRACE_RED";
        result.cases.push(traceEvidenceErrorReceipt(fixture.id, sha256(fixture.query), Number((performance.now() - started).toFixed(3)), evidenceFailure));
        continue;
      }
      result.cases.push({
        status: "captured",
        fixture_id: fixture.id,
        query_sha256: sha256(fixture.query),
        fixture_namespace: fixture.namespace,
        isolation: fixture.isolation === true,
        expected_keys: expectedKeys,
        forbidden_keys: forbiddenKeys,
        elapsed_ms: Number((performance.now() - started).toFixed(3)),
        span_durations_ms: {
          vector: duration(vectorSpan), lexical: duration(lexicalSpan), rrf: duration(rankSpan), execute: duration(executeSpan),
        },
        vector_arm: { ...receiptEvidence(vectorEvidence), count: vectorRows.length, rows: vectorRows, expected: expectedKeys.map((key) => targetEvidence(vectorRows, key)) },
        lexical_arm: { ...receiptEvidence(lexicalEvidence), count: lexicalRows.length, rows: lexicalRows, expected: expectedKeys.map((key) => targetEvidence(lexicalRows, key)) },
        rrf_fusion: { ...receiptEvidence(rankEvidence), candidate_count: fusedRows.length, selected_count: selectedRows.length, rows: fusedRows, expected: expectedKeys.map((key) => targetEvidence(fusedRows, key)) },
        selected_top10: { ...receiptEvidence(executeEvidence), count: actualRows.length, rows: actualRows, expected: expectedKeys.map((key) => targetEvidence(actualRows, key)), forbidden_present: forbiddenKeys.filter((key) => allIds.includes(key)), matches_rrf_choice: allIds.join("\n") === selectedRows.map((row) => row.id).join("\n") },
        trace_span_names: Array.isArray(body.spans) ? body.spans.map((span) => span.name) : [],
      });
    } catch (error) {
      result.status = "TRACE_RED";
      result.cases.push({ status: "ERROR", fixture_id: fixture.id, query_sha256: sha256(fixture.query), elapsed_ms: Number((performance.now() - started).toFixed(3)), error_class: error instanceof Error ? error.name : "unknown" });
    }
  }
} finally {
  await pool.end();
}
if (result.cases.some((item) => item.status !== "captured")) {
  result.status = "TRACE_RED";
}
writeOutput(result);
process.stdout.write(`AB_TRACE_${result.status} variant=${variant} cases=${result.cases.length}\n`);
process.exitCode = result.status === "captured" ? 0 : 1;
