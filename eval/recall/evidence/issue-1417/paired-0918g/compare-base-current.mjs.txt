#!/usr/bin/env node
import { createHash } from "node:crypto";
import { readFileSync, writeFileSync } from "node:fs";

const EXPECTED = {
  base: {
    file: "c27c98af915e2a96b8d13beb2a0e1a56577cf5717462b05b88502bdca78341cf",
    source: "6ecd002aa6ee86a55c44f103d5e4805a1481f7df",
    archive: "18e6acffaef9e43e907f544e5a574721b486164f48b4810a92077c09b15e53bb",
    backup_sha256: "0eed3412ef3b065616908675ea2b25d9363f6091a874679a5e3f13ddf72003c2",
  },
  current: {
    file: "677dd83a7f1e542fa9e5e5c3e03228afb95f5a679f13ebe53bef5735161c6920",
    source: "4eb1810316c68b711df9b6cbeec68c50db73f2ee",
    archive: "bb91c93dcb22b5f0825ab4b8eacf97563ec2a643",
    backup_sha256: "0eed3412ef3b065616908675ea2b25d9363f6091a874679a5e3f13ddf72003c2",
  },
  fixture: "b614dd7d72e54bc7faf1ec97f2a72f5edfe67238825c934704c9930048439a7a",
  cache: "3dead372e3c079f968979908c9bb5434fa68298b1cf75814de30f706a1ae1699",
  backup: "0eed3412ef3b065616908675ea2b25d9363f6091a874679a5e3f13ddf72003c2",
};

const FOCUS = new Map([
  ["dirty-paths-are-owned", "session_event:75d29b27-2a3c-400e-b6a2-7c3c218e4681"],
  ["wave-sizing-for-head-sessions", "session_event:71da1b6f-d213-4514-960a-825ee44e37fb"],
]);
const ARMS = ["vector_arm", "lexical_arm", "rrf_fusion", "selected_top10"];

function hash(buffer) {
  return createHash("sha256").update(buffer).digest("hex");
}
function rows(item, arm) {
  const value = item?.[arm]?.rows;
  return Array.isArray(value) ? value : [];
}
function rank(item, key) {
  return rows(item, "selected_top10").find((row) => row?.id === key)?.rank ?? null;
}
function namespacesMatch(item) {
  return ARMS.every((arm) => rows(item, arm).every((row) => row?.namespace === item.fixture_namespace));
}
function isolationEvidenceNonempty(item) {
  if (item?.isolation !== true) return true;
  const complete = ARMS.every((arm) => {
    const evidence = item?.[arm];
    const declared = arm === "rrf_fusion" ? evidence?.candidate_count : evidence?.count;
    return Number.isInteger(declared) && declared >= 0 && declared === rows(item, arm).length;
  });
  return complete && ARMS.some((arm) => rows(item, arm).length > 0);
}
function forbidden(item) {
  const keys = new Set(Array.isArray(item?.forbidden_keys) ? item.forbidden_keys : []);
  return ARMS.flatMap((arm) => rows(item, arm).filter((row) => keys.has(row?.id)).map((row) => ({ arm, id: row.id })));
}
function expectedPresent(item) {
  const keys = Array.isArray(item?.expected_keys) ? item.expected_keys : [];
  return keys.filter((key) => rank(item, key) !== null);
}
function passed(item) {
  if (item?.isolation === true) return isolationEvidenceNonempty(item) && namespacesMatch(item) && forbidden(item).length === 0;
  return expectedPresent(item).length > 0;
}
function vectorDiff(base, current) {
  const left = rows(base, "vector_arm");
  const right = rows(current, "vector_arm");
  const count = Math.max(left.length, right.length);
  const changes = [];
  for (let index = 0; index < count; index += 1) {
    const a = left[index] ?? null;
    const b = right[index] ?? null;
    if (a?.id === b?.id && a?.namespace === b?.namespace && Object.is(a?.score, b?.score)) continue;
    changes.push({ index, base: a && { id: a.id, namespace: a.namespace, score: a.score }, current: b && { id: b.id, namespace: b.namespace, score: b.score } });
  }
  return changes;
}
function validateDocument(document, expected, contract, label) {
  const errors = [];
  if (document.status !== "captured") errors.push(`${label}:status`);
  if (document.source_head !== expected.source) errors.push(`${label}:source`);
  if (document.source_archive_sha256 !== expected.archive) errors.push(`${label}:archive`);
  if (document.fixture_sha256 !== contract.fixture || document.fixture_count !== 24) errors.push(`${label}:fixture`);
  if (document.cache_sha256 !== contract.cache) errors.push(`${label}:cache`);
  if (expected.backup_sha256 !== contract.backup) errors.push(`${label}:backup`);
  if (document.mode !== "hybrid" || document.limit !== 10 || document.offset !== 0) errors.push(`${label}:search_contract`);
  if (document.embedding_dimensions !== 768) errors.push(`${label}:dimensions`);
  if ((document.cases ?? []).length !== 24 || new Set((document.cases ?? []).map((item) => item.fixture_id)).size !== 24) errors.push(`${label}:cases`);
  if ((document.cases ?? []).some((item) => item.status !== "captured")) errors.push(`${label}:case_status`);
  if (Object.values(document.model_calls ?? {}).some((value) => value !== 0)) errors.push(`${label}:model_calls`);
  return errors;
}

export function compare(base, current, pins = EXPECTED) {
  const errors = [...validateDocument(base, pins.base, pins, "base"), ...validateDocument(current, pins.current, pins, "current")];
  const baseById = new Map(base.cases.map((item) => [item.fixture_id, item]));
  const currentById = new Map(current.cases.map((item) => [item.fixture_id, item]));
  const cases = [...new Set([...baseById.keys(), ...currentById.keys()])].sort().map((fixtureId) => {
    const before = baseById.get(fixtureId);
    const after = currentById.get(fixtureId);
    const changes = vectorDiff(before, after);
    return {
      fixture_id: fixtureId,
      base_pass: passed(before),
      current_pass: passed(after),
      pass_to_miss: passed(before) && !passed(after),
      base_expected_ranks: (before?.expected_keys ?? []).map((key) => ({ key, rank: rank(before, key) })),
      current_expected_ranks: (after?.expected_keys ?? []).map((key) => ({ key, rank: rank(after, key) })),
      base_namespace_ok: namespacesMatch(before),
      current_namespace_ok: namespacesMatch(after),
      base_forbidden: forbidden(before),
      current_forbidden: forbidden(after),
      base_isolation_evidence_nonempty: isolationEvidenceNonempty(before),
      current_isolation_evidence_nonempty: isolationEvidenceNonempty(after),
      vector_equal: changes.length === 0,
      vector_changes: changes,
    };
  });
  const focus = [...FOCUS].map(([fixtureId, key]) => ({ fixture_id: fixtureId, key, rank: rank(currentById.get(fixtureId), key) }));
  const acceptance = {
    all_24_present: cases.length === 24,
    no_pass_to_miss: cases.every((item) => !item.pass_to_miss),
    vector_parity: cases.every((item) => item.vector_equal),
    namespaces_bound: cases.every((item) => item.base_namespace_ok && item.current_namespace_ok),
    no_forbidden: cases.every((item) => item.base_forbidden.length === 0 && item.current_forbidden.length === 0),
    isolation_evidence_nonempty: cases.every((item) => item.base_isolation_evidence_nonempty && item.current_isolation_evidence_nonempty),
    focus_within_10: focus.every((item) => item.rank !== null && item.rank <= 10),
  };
  const pass = errors.length === 0 && Object.values(acceptance).every(Boolean);
  return { schema: "openbrain.recall_base_current_comparison.v1", status: pass ? "PASS" : "RECALL_RED", pins: { ...pins, base_file: pins.base.file, current_file: pins.current.file }, errors, acceptance, focus, cases };
}

function main() {
  const [basePath, currentPath, outputPath] = process.argv.slice(2);
  if (!basePath || !currentPath || !outputPath) throw new Error("usage: compare-base-current.mjs BASE CURRENT OUTPUT");
  const baseBuffer = readFileSync(basePath);
  const currentBuffer = readFileSync(currentPath);
  if (hash(baseBuffer) !== EXPECTED.base.file) throw new Error("base file hash mismatch");
  if (hash(currentBuffer) !== EXPECTED.current.file) throw new Error("current file hash mismatch");
  const output = compare(JSON.parse(baseBuffer), JSON.parse(currentBuffer));
  writeFileSync(outputPath, `${JSON.stringify(output, null, 2)}\n`);
  console.log(`BASE_CURRENT_${output.status} cases=${output.cases.length}`);
  process.exitCode = output.status === "PASS" ? 0 : 1;
}

if (import.meta.url === `file://${process.argv[1]}`) main();
