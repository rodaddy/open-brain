function record(value) {
  return value !== null && typeof value === "object" && !Array.isArray(value)
    ? value
    : {};
}

function hasIdentity(candidate) {
  return (
    (typeof candidate.row_id === "string" ||
      typeof candidate.id === "string" ||
      typeof candidate.path === "string") &&
    typeof candidate.source_type === "string" &&
    typeof candidate.namespace === "string"
  );
}

function hasScore(candidate, scoreKeys) {
  return scoreKeys.some(
    (key) => typeof candidate[key] === "number" && Number.isFinite(candidate[key]),
  );
}

export function readTraceEvidence(span, scoreKeys) {
  const spanPresent = span !== null && typeof span === "object" && !Array.isArray(span);
  const metadata = record(span?.metadata);
  const output = record(span?.output);
  const degraded = metadata.payload_degraded === true;
  const reason = degraded ? metadata.payload_degradation_reason ?? null : null;
  const omittedSpans =
    typeof metadata.additional_spans_omitted === "number"
      ? metadata.additional_spans_omitted
      : 0;
  const explicitlyTruncated =
    metadata.payload_evidence_truncated === true || output.evidence_truncated === true;
  const evidenceComplete = degraded
    ? output.evidence_complete === true && !explicitlyTruncated && omittedSpans === 0
    : !explicitlyTruncated && omittedSpans === 0;
  const source = degraded && evidenceComplete ? record(output.evidence) : output;
  const candidates = Array.isArray(source.candidates) ? source.candidates : [];
  const declaredCountValue =
    typeof source.count === "number"
      ? source.count
      : typeof source.candidate_count === "number"
        ? source.candidate_count
        : null;
  const declaredCount =
    Number.isInteger(declaredCountValue) && declaredCountValue >= 0
      ? declaredCountValue
      : null;
  const candidateShapeComplete = candidates.every((value) => {
    const candidate = record(value);
    return hasIdentity(candidate) && hasScore(candidate, scoreKeys);
  });
  const countComplete = declaredCount !== null && declaredCount === candidates.length;
  const usable = spanPresent && evidenceComplete && candidateShapeComplete && countComplete;
  return {
    candidates: usable ? candidates : [],
    declared_count: declaredCount,
    evidence_complete: evidenceComplete,
    payload_degraded: degraded,
    payload_degradation_reason: reason,
    additional_spans_omitted: omittedSpans,
    payload_evidence_truncated: explicitlyTruncated,
    span_present: spanPresent,
    candidate_shape_complete: candidateShapeComplete,
    count_complete: countComplete,
    usable,
  };
}

export function isolationEvidencePresent(rows) {
  return Array.isArray(rows) && rows.length > 0;
}

export function traceCaseStatus(evidenceByArm, isolation, selectedRows) {
  return traceCaseFailure(evidenceByArm, isolation, selectedRows) === null
    ? "captured"
    : "TRACE_RED";
}

export function traceCaseFailure(evidenceByArm, isolation, selectedRows) {
  const unusable_arms = Object.entries(evidenceByArm)
    .filter(([, evidence]) => !evidence.usable)
    .map(([arm, evidence]) => ({
      arm,
      span_present: evidence.span_present,
      evidence_complete: evidence.evidence_complete,
      candidate_shape_complete: evidence.candidate_shape_complete,
      count_complete: evidence.count_complete,
      declared_count: evidence.declared_count,
      payload_degraded: evidence.payload_degraded,
      payload_degradation_reason: evidence.payload_degradation_reason,
      additional_spans_omitted: evidence.additional_spans_omitted,
      payload_evidence_truncated: evidence.payload_evidence_truncated,
    }));
  if (unusable_arms.length > 0) return { reason: "incomplete_trace_evidence", unusable_arms };
  if (isolation && !isolationEvidencePresent(selectedRows))
    return { reason: "empty_isolation_evidence", unusable_arms: [] };
  return null;
}

export function traceEvidenceErrorReceipt(fixture_id, query_sha256, elapsed_ms, failure) {
  return {
    status: "ERROR",
    fixture_id,
    query_sha256,
    elapsed_ms,
    error_class: "TraceEvidenceError",
    trace_evidence_failure: failure,
  };
}
